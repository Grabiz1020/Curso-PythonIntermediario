Models
    Serve para setar como vai ser organizado banco de dados por meio de classes ou "Models"
        Exemplo de  Model
    class book(models.Model):
        index = models.IntegerField("id", max_digits=20)
        class covercolors(models.TextChoices):
            BLUE = "blue", "Blue"
            GREEN = "green", "Green"
            RED = "red", "Red"
        title = models.CharField("title", max_length=160)
        author = models.CharField("author", max_length=70)
        slug = models.SlugField(unique=True)
        description =  models.TextField("description")
        price = models.DecimalField("price", max_digits=8)
        cover_imageurl = models.URLField(
            "image url",
            max_length=500,
            blank=True,
            help_text="Insert URL of the Image. If empty, use coverstyle.",
        )
        cover_style = models.CharField(
            "cover style",
            max_length=12,
            choice=covercolors.choices,
            default=covercolors.BLUE,
        )
